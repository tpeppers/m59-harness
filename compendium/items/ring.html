RevealHiddenAttributes()

{
      local i, bDone;

      bDone = FALSE;
      
      
      for i in plItem_Attributes
      {
         if NOT Send(self, @GetIDStatusFromCompound, #compound=First(i))
         {
            SetNth(i, 1, (First(i) + 1));
            bDone = TRUE;
         }
      }
      
      if vrName <> vrRealName
      {
         vrName = vrRealName;
         vrDesc = vrRealDesc;
         bDone = TRUE;
      }
      
      Send(self,@RevealHiddenColor);      
      
      return bDone;
   }
