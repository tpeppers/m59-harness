CanPayCosts(who = $, lTargets = $)

{
      local oRoom;

      oRoom = send(who,@GetOwner);

      
      if NOT send(oRoom,@AllowGuildAttack,#what=who)
         AND Send(SYS,@IsPKAllowed)
         AND NOT IsClass(who,&DM)
      {
         
         Send(who,@MsgSendUser,#message_rsc=Spellbane_no_towns);

         return FALSE;
      }

      propagate;
   }
