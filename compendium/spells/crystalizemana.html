This song catches the mana expended in singing, crystalizing it into a physical form. The mana crystal stores the mana for later use, in essence allowing the singer to store extra mana. Only one crystal at a time can be created by a singer, but the longer this song is sung, the more mana that is stored in the crystal. Since the mana is attuned to the singer, the crystal cannot be shared with others. This song requires 1 sapphire and 1 uncut seraphym to sing.

kod/object/passive/spell/jala/crysmana.kod:62

StartPeriodicEnchantment(where = $, state = $)

{
      local oCaster, iSpellpower, oCrystal, iAmount;

      
      oCaster = Nth(state,2);

      
      iSpellpower = Nth(state,3);

      
      oCrystal = Send(oCaster,@FindHolding,#class=&ManaCrystal);

      if oCrystal = $
      {
         oCrystal = Create(&ManaCrystal,#oCaster=oCaster);
         if Send(oCaster,@ReqNewHold,#what=oCrystal)
         {
            Send(oCaster,@NewHold,#what=oCrystal);
         }
         else
         {
            
            Send(oCrystal,@Delete);
            Send(oCaster,@MsgSendUser,#message_rsc=CrystalizeMana_no_room);

            
            Post(oCaster,@BreakTrance,#event=EVENT_STEER);

            propagate;
         }
      }

      
      
      iAmount = 1;

      
      if random(1,99) <= iSpellpower
      {
         iAmount = iAmount + 1;
      }

      Send(oCrystal,@IncreaseMana,#amount=iAmount);

      propagate;
   }
