kod/object/passive/spell/atakspel/vampdrn.kod:66

DoSideEffect(who = $, victim = $, damage = 0)

{
      local iToHeal;
      
      if who = $ OR victim = $
      {
         propagate;
      }

      
      if damage <> $
      {
         
         iToHeal = bound(damage/DAMAGE_FACTOR_TO_HEAL,1,$);
      }
      else
      {
         
         iToHeal = bound(piDamageMax/DAMAGE_FACTOR_TO_HEAL,1,$);
         Post(who,@MsgSendUser,#message_rsc=VampiricDrain_death,
              #parm1=Send(victim,@GetDef),#parm2=Send(victim,@GetName));
      }

      Send(who,@GainHealth,#amount=iToHeal);

      propagate;
   }
