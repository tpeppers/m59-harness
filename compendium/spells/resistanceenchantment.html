InitializeEnchantmentState(who = $, iSpellPower = 0, target = $)

{
      local iValue;
      
      iValue = send(self,@GetResistanceStrength,#iSpellPower=iSpellPower);
      send(self,@AddResistances,#who=target,#value=iValue);
      
      return iValue;
   }
