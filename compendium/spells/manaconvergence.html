ModifySpellPower(oRoom = $, oSpell = $)

{
      local iChange;

      iChange = 0;

      if oRoom <> $
      {
         iChange = send(oRoom,@GetEnchantedState,#what=self);
         if iChange <> $
         {
            iChange = nth(iChange,3);
         }
      }

      
      if iChange = $
         OR IsClass(oSpell,&AntiMagicAura)
      {
         iChange = 0;
      }

      
      return iChange;
   }
