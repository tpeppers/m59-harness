The powerful smell of rotted flesh and embalming herbs assault your nostrils. This shambling horror in front of you seems like it was crafted in madness, and birthed in nightmares. Various limbs, none of them matching, sprout from this monster in disarray, and in its eye a strange malignance glows. Run for your life.

The reference character is a mid-game character — 130 max health, dodge 50, parry 40, agility 40, stroke 50, proficiency 50, aim 40, giving defence 585 and offence 605. Those are the same on every page in this bestiary, so the percentages are directly comparable between creatures. Armour is not included — see Armour for what it subtracts.

It attacks with bludgeon damage. Armour resists by type, so what you wear against this creature matters more than how much you wear.

What killing it does to your karma

Its karma is -75, and a kill is treated as an act whose karma is the negative of the victim's — so killing this creature pushes you towards good.

base = your_karma − (−victim_karma) = your_karma − 75 change = −( base³ / 2500 + 5 × base ) / (11 − swing) % hundredths of a karma point bounded to ±150 for a monster kill — at most 1.5 karma per kill

Decides which shield/weapon/drumstick is part of the thrasher and sets the appropriate overlay and stat adjustments.

{
      local RandomNumber, oObject;

      
      RandomNumber = random(0,2);
      if RandomNumber = 0
      {
         
         viLevel = viLevel + 10;
         oObject = Create(&GoldShield);
         Send(self,@NewHold,#what=oObject);
         prShieldOverlay = thrasher_shield_overlay;
      }
      else
      {
         if RandomNumber = 1
         {
            
            oObject = Create(&Mace);
            Send(self,@NewHold,#what=oObject);
            prShieldOverlay = thrasher_mace_overlay;
         }
         else
         {
            
            viDifficulty = viDifficulty + 1;
            oObject = Create(&Scimitar);
            Send(self,@NewHold,#what=oObject);
            prShieldOverlay = thrasher_scimitar_overlay;
         }
      }

      return;
   }

GetFinalTargets()

{
      local oRoom, i, each_obj, lValidTargets;

      lValidTargets = $;

      oRoom = Send(self,@GetOwner);
      for i in Send(oRoom,@GetHolderActive)
      {
         each_obj = Send(oRoom,@HolderExtractObject,#data=i);
         if IsClass(each_obj,&User)
            AND Send(self,@CanReach,#what=each_obj)
         {
            lValidTargets = Cons(each_obj,lValidTargets);
         }
      }

      return lValidTargets; 
   }

DoSpinAttack(lFinalTargets = $)

{
      local i, lTargets, iDamage;

      iDamage = Random(SPIN_DAMAGE_MIN,SPIN_DAMAGE_MAX);

      lTargets = lFinalTargets;
      
      if lTargets = $
      {
         lTargets = Send(self,@GetFinalTargets);
      }

      for i in lTargets
      {
         iDamage = Send(i,@AssessDamage,#what=self,#damage=iDamage,
                        #atype=viAttack_type,#aspell=viAttack_spell);
         if iDamage = $
         {
            Send(self,@KilledSomething,#what=i,#use_weapon=self);
         } 

         Send(i,@MsgSendUser,#message_rsc=thrasher_spin_message);
         Send(i,@WaveSendUser,#wave_rsc=thrasher_sound_attack);
      }

      return;
   }

kod/object/active/holder/nomoveon/battler/monster/thrasher.kod:372

SpinTimer()

{
      ptSpin = $;

      piAnimation = ANIM_SPIN;

      if piSpinAngle > MAX_ANGLE
      {
         
         piAnimation = ANIM_NONE;
         Send(poOwner,@SomethingChanged,#what=self);

         
         Send(self,@DoSpinAttack);

         return;
      }

      Send(poOwner,@SomethingTurned,#what=self,#new_angle=piSpinAngle);
      Send(poOwner,@SomethingChanged,#what=self);
      piSpinAngle = piSpinAngle + (MAX_ANGLE / 4);

      ptSpin = CreateTimer(self,@SpinTimer,piSpinDelay);

      return;   
   }
