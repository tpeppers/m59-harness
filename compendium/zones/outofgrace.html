RoomReqCommunication(who = $, type = $, string = $, users = $)

{
       local i;

       if IsClass(who,&DM)
       {  return TRUE;  }

       if type = SAY_GUILD or type = SAY_EVERYONE or type = SAY_YELL
       {
          Send(who,@MsgSendUser,#message_rsc=room_no_sound);

          return FALSE;
       }

       if type = SAY_GROUP or type = SAY_GROUP_ONE
       {
          for i in users
          {
             if not isClass(i,&DM)
             {
                Send(who,@MsgSendUser,#message_rsc=room_no_sound);

                return FALSE;
             }
          }
       }

       return TRUE;
   }
