OkayToGetChalice()

{
      local i, Each_obj;

      
      if ptIllusionResetTimer = $
      {
         return FALSE;
      }

      for i in plActive
      {
         each_obj=Send(self,@holderExtractObject,#data=i);
         if isClass(each_obj,&orc)
         {
            return FALSE;
         }
      }

      return TRUE;
   }
