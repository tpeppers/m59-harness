ActivatePortal()

{
      
      if poNode <> $
      {
         return;
      }

      poNode = Send(SYS,@FindNodeByNum,#num=NODE_Q);
      Send(poNode,@NodeAppear,#where=self,#row=57,#col=45,
           #fine_row=32,#fine_col=32);         

      return;
   }

DeactivatePortal()

{

   }
